CONSTANTS:
    // collection of provinces
    PROVINCE
    // max number of availabilities to show
    PRESCRIPTION_SLOTS_SHOWN
    // time after which the last notified availability expires (1 week)
    PRESCRIPTION_NOTIFICATION_RESET

STATE:
    // each one stores: id, name, chat, earliest, booking (date, time, location), notified (date, timestamp), last_error
    monitored_prescriptions


// helper function
// a single search of the prescription returns its name, its booking (if any) and the availabilities
FUNCTION check(prescription_id):
    prescription = search prescription_id on the booking website
    // cancelled appointments are ignored, with more than one the earliest is kept
    booking = earliest not cancelled appointment of prescription, or NONE
    availabilities = []
    FOR EACH p IN PROVINCE:
        availabilities.append(all availabilities in p related to prescription_id)
    SORT availabilities BY date ASCENDING
    RETURN prescription.name, availabilities, booking


// helper function
// with a booking only the availabilities earlier than it are worth notifying
FUNCTION notifiable(availabilities, booking):
    IF booking is NONE:
        RETURN availabilities
    RETURN availabilities WHERE a.date < booking.date


// helper function
FUNCTION should_notify(notified, availabilities):
    IF availabilities is empty:
        RETURN FALSE
    IF notified is NONE OR now - notified.timestamp >= PRESCRIPTION_NOTIFICATION_RESET:
        RETURN TRUE
    RETURN availabilities[0].date < notified.date


// check_prescription
COMMAND check_prescription(prescription_id):
    IF prescription_id is not valid:
        send_message("Invalid prescription code")
        RETURN
    name, availabilities, booking = check(prescription_id)
    send_message(name, prescription_id)
    IF booking is not NONE:
        send_message(booking.date, booking.time, booking.location)
    IF availabilities is empty:
        send_message("No availability right now")
    FOR EACH a IN first PRESCRIPTION_SLOTS_SHOWN items of availabilities:
        send_message(a.date, a.time, a.location)


// monitor_prescription
COMMAND monitor_prescription(prescription_id):
    IF prescription_id is not valid:
        send_message("Invalid prescription code")
    ELSE IF prescription_id IN monitored_prescriptions:
        send_message("This prescription is already being monitored")
    ELSE:
        // same output as check_prescription
        name, availabilities, booking = check_prescription(prescription_id)
        candidates = notifiable(availabilities, booking)
        // what was just shown counts as notified
        notified = (candidates[0].date, now) IF candidates is not empty ELSE NONE
        monitored_prescriptions.add(prescription_id, name, chat, booking, notified)
        send_message("Monitoring started")


// stop_prescription
COMMAND stop_prescription(prescription_id):
    IF prescription_id NOT IN monitored_prescriptions:
        send_message("This prescription is not being monitored")
    ELSE:
        monitored_prescriptions.remove(prescription_id)
        send_message("Monitoring stopped")


// prescriptions
COMMAND prescriptions():
    IF monitored_prescriptions is empty:
        send_message("No prescription is being monitored")
    FOR EACH mp IN monitored_prescriptions:
        message = mp.name + " " + mp.id + " " + mp.earliest
        IF mp.booking is not NONE:
            message += mp.booking.date + " " + mp.booking.time
        send_message(message)


// periodic_prescription_check
JOB periodic_prescription_check():
    FOR EACH mp IN monitored_prescriptions:
        TRY:
            name, availabilities, booking = check(mp.id)
        CATCH error:
            // errors are notified only once, until they change
            IF error != mp.last_error:
                mp.last_error = error
                send_message(mp.id, error)
            CONTINUE

        // the appointment might disappear from the prescription after the visit, so the old booking counts too
        visit = booking OR mp.booking
        IF visit is not NONE AND visit.date <= now:
            monitored_prescriptions.remove(mp.id)
            send_message("Visit done, monitoring stopped")
            CONTINUE

        // booked, moved or cancelled since the last check
        IF booking != mp.booking:
            send_message("Booked" / "Booking changed" / "Booking cancelled")
            // the last notified availability was compared with the old booking, start over
            mp.notified = NONE

        candidates = notifiable(availabilities, booking)
        IF should_notify(mp.notified, candidates):
            mp.notified = (candidates[0].date, now)
            send_message("An earlier slot became available!")
            // title and booking are sent once, before the availabilities
            send_message(mp.name, mp.id)
            IF booking is not NONE:
                send_message(booking.date, booking.time, booking.location)
            FOR EACH a IN first PRESCRIPTION_SLOTS_SHOWN items of candidates:
                send_message(a.date, a.time, a.location)
        ELSE IF booking != mp.booking:
            send_message(mp.name, mp.id)
            IF booking is not NONE:
                send_message(booking.date, booking.time, booking.location)

        mp.name = name
        mp.earliest = availabilities[0].date IF availabilities is not empty ELSE NONE
        mp.booking = booking
        mp.last_error = NONE
